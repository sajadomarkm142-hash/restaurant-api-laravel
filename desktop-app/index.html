const { app, BrowserWindow } = require('electron'); const { spawn } = require('child_process'); const fs = require('fs'); const path = require('path'); let backendProcess = null; let frontendProcess = null; let splashWindow = null; let mainWindow = null; function startBackend() { const backendPath = path.join(__dirname, '..', 'restaurant-api-laravel'); if (!fs.existsSync(path.join(backendPath, 'artisan'))) { return; } backendProcess = spawn('php', ['artisan', 'serve', '--host', '127.0.0.1', '--port', '8000'], { cwd: backendPath, detached: true, stdio: 'ignore', windowsHide: true, shell: false }); backendProcess.unref(); } function startFrontend() { const frontendPath = path.join(__dirname, '..', 'restaurant-frontend'); if (!fs.existsSync(path.join(frontendPath, 'package.json'))) { return; } const builtFrontend = path.join(frontendPath, 'dist', 'index.html'); if (fs.existsSync(builtFrontend)) { return; } const npmCommand = process.platform === 'win32' ? 'npm.cmd' : 'npm'; frontendProcess = spawn( npmCommand, ['run', 'dev', '--', '--host', '127.0.0.1', '--port', '5173'], { cwd: frontendPath, detached: true, stdio: 'ignore', windowsHide: true, shell: false } ); frontendProcess.unref(); } function createSplashWindow() { splashWindow = new BrowserWindow({ width: 820, height: 520, frame: false, transparent: true, resizable: false, movable: false, alwaysOnTop: true, backgroundColor: '#0f172a', show: true, webPreferences: { contextIsolation: true, nodeIntegration: false, enableRemoteModule: false } }); splashWindow.loadFile(path.join(__dirname, 'index.html')); } function createMainWindow() { mainWindow = new BrowserWindow({ width: 1400, height: 900, minWidth: 1100, minHeight: 700, backgroundColor: '#0f172a', autoHideMenuBar: true, show: false, title: 'Restaurant POS', webPreferences: { preload: path.join(__dirname, 'preload.js'), contextIsolation: true, nodeIntegration: false, enableRemoteModule: false } }); mainWindow.setMenuBarVisibility(false); const builtFrontend = path.join(__dirname, '..', 'restaurant-frontend', 'dist', 'index.html'); if (fs.existsSync(builtFrontend)) { mainWindow.loadFile(builtFrontend); } else { mainWindow.loadURL('http://127.0.0.1:5173'); } mainWindow.webContents.setWindowOpenHandler(() => ({ action: 'deny' })); mainWindow.once('ready-to-show', () => { mainWindow.show(); }); if (splashWindow) { splashWindow.close(); splashWindow = null; } } app.whenReady().then(() => { startBackend(); startFrontend(); createSplashWindow(); setTimeout(() => { createMainWindow(); }, 1800); app.on('activate', () => { if (BrowserWindow.getAllWindows().length === 0) { createMainWindow(); } }); }); app.on('window-all-closed', () => { if (backendProcess) { try { backendProcess.kill(); } catch (error) {} } if (frontendProcess) { try { frontendProcess.kill(); } catch (error) {} } if (process.platform !== 'darwin') { app.quit(); } });
